    59→  name: string

    65→type ProductRow = {

    67→  name: string
    68→  slug: string
    69→  status: string
    70→  price: number
    71→  compareAt?: number | null
    72→  featured: boolean
    73→  sku?: string | null

    97→  featured: boolean
    98→  categoryId: string
    99→  tags: string
   100→  images: ImageDraft[]
   101→  colors: ColorDraft[]
   102→  sizes: SizeDraft[]
   103→}

   115→  tags: '',
   116→  images: [],
   117→  colors: [],
   118→  sizes: [],
   119→}

   133→export function AdminProducts({ settings }: { settings: SiteSettings }) {
   134→  const qc = useQueryClient()
   135→  const [open, setOpen] = useState(false)
   136→  const [editId, setEditId] = useState<string | null>(null)
   137→  const [form, setForm] = useState<ProductForm>(EMPTY_FORM)
   138→  const [deleteId, setDeleteId] = useState<string | null>(null)
   139→  const [deleteName, setDeleteName] = useState('')
   140→  const [search, setSearch] = useState('')
   141→
   142→  // Lista de produtos (admin)
   143→  const productsQuery = useQuery<{ items: ProductRow[] }>({
   144→    queryKey: ['admin', 'products'],
   145→    queryFn: () =>
   146→      fetch('/api/admin/products', { cache: 'no-store' }).then((r) =>
   147→        r.ok ? r.json() : Promise.reject(new Error('products')),
   148→      ),
   149→    staleTime: 15_000,
   150→  })
   151→
   152→  // Lista de categorias (admin) — inclui inativas
   153→  const categoriesQuery = useQuery<{ items: Category[] }>({
   154→    queryKey: ['admin', 'categories'],
   155→    queryFn: () =>
   156→      fetch('/api/admin/categories', { cache: 'no-store' }).then((r) =>
   157→        r.ok ? r.json() : Promise.reject(new Error('categories')),
   158→      ),
   159→    staleTime: 30_000,
   160→  })
   161→
   162→  const categories = categoriesQuery.data?.items ?? []
   163→
   164→  // Mutation salvar
   165→  const saveMutation = useMutation({
   166→    mutationFn: async () => {
   167→      const body = {
   168→        name: form.name.trim(),
   169→        description: form.description.trim(),
   170→        price: Number(form.price) || 0,
   171→        compareAt: form.compareAt ? Number(form.compareAt) : null,
   172→        sku: form.sku.trim() || null,
   173→        status: form.status,
   174→        featured: form.featured,
   175→        categoryId: form.categoryId,
   176→        tags: form.tags.trim() || null,
   177→        images: form.images.map((im, i) => ({
   178→          url: im.url,
   179→          isMain: i === 0 ? true : im.isMain,
   180→          order: i,
   181→        })),
   182→        colors: form.colors
   183→          .filter((c) => c.name.trim() && c.hex)
   184→          .map((c) => ({
   185→            name: c.name.trim(),
   186→            hex: c.hex,
   187→            stock: Number(c.stock) || 0,
   188→          })),
   189→        sizes: form.sizes
   190→          .filter((s) => s.name.trim())
   191→          .map((s) => ({
   192→            name: s.name.trim(),
   193→            stock: Number(s.stock) || 0,
   194→          })),
   195→      }
   196→      const url = editId
   197→        ? `/api/admin/products/${editId}`
   198→        : '/api/admin/products'
   199→      const method = editId ? 'PUT' : 'POST'
   200→      const r = await fetch(url, {
   201→        method,
   202→        headers: { 'content-type': 'application/json' },
   203→        body: JSON.stringify(body),
   204→      })
   205→      const data = await r.json()
   206→      if (!r.ok) throw new Error(data?.error ?? 'Falha ao salvar')
   207→      return data
   208→    },
   209→    onSuccess: () => {
   210→      toast.success(editId ? 'Produto atualizado ✨' : 'Produto criado ✨')
   211→      qc.invalidateQueries({ queryKey: ['admin', 'products'] })
   212→      setOpen(false)
   213→    },
   214→    onError: (e: Error) => toast.error(e.message),
   215→  })
   216→
   217→  // Mutation delete
   218→  const deleteMutation = useMutation({
   219→    mutationFn: async (id: string) => {
   220→      const r = await fetch(`/api/admin/products/${id}`, {
   221→        method: 'DELETE',
   222→      })
   223→      const data = await r.json()
   224→      if (!r.ok) throw new Error(data?.error ?? 'Falha ao excluir')
   225→      return data
   226→    },
   227→    onSuccess: () => {
   228→      toast.success('Produto excluído.')
   229→      setDeleteId(null)
   230→      qc.invalidateQueries({ queryKey: ['admin', 'products'] })
   231→    },
   232→    onError: (e: Error) => toast.error(e.message),
   233→  })
   234→
   235→  // Abrir editor para NOVO
   236→  function newProduct() {
   237→    setEditId(null)
   238→    setForm({
   239→      ...EMPTY_FORM,
   240→      categoryId: categories[0]?.id ?? '',
   241→    })
   242→    setOpen(true)
   243→  }
   244→
   245→  // Abrir editor para EDITAR — busca detalhes via /api/products/[slug]
   246→  async function editProduct(p: ProductRow) {
   247→    setEditId(p.id)
   248→    setForm({
   249→      name: p.name ?? '',
   250→      slug: p.slug ?? '',
   251→      description: p.description ?? '',
   252→      price: String(p.price ?? ''),
   253→      compareAt: p.compareAt != null ? String(p.compareAt) : '',
   254→      sku: p.sku ?? '',
   255→      status: (p.status as ProductForm['status']) ?? 'active',
   256→      featured: !!p.featured,
   257→      categoryId: p.categoryId,
   258→      tags: p.tags ?? '',
   259→      images: (p.images ?? []).map((i) => ({
   260→        url: i.url,
   261→        isMain: i.isMain,
   262→        order: i.order,
   263→      })),
   264→      colors: (p.colors ?? []).map((c) => ({
   265→        name: c.name,
   266→        hex: c.hex,
   267→        stock: c.stock,
   268→      })),
   269→      sizes: (p.sizes ?? []).map((s) => ({
   270→        name: s.name,
   271→        stock: s.stock,
   272→      })),
   273→    })
   274→    // Se não vieram images/colors/sizes do list, busca o detalhe público:
   275→    if (!p.images && !p.colors && !p.sizes) {
   276→      try {
   277→        const r = await fetch(`/api/products/${p.slug}`, {
   278→          cache: 'no-store',
   279→        })
   280→        if (r.ok) {
   281→          const data = await r.json()
   282→          const prod = data.product
   283→          if (prod) {
   284→            setForm((f) => ({
   285→              ...f,
   286→              description: prod.description ?? f.description,
   287→              images: (prod.images ?? []).map((i: any) => ({
   288→                url: i.url,
   289→                isMain: i.isMain,
   290→                order: i.order ?? 0,
   291→              })),
   292→              colors: (prod.colors ?? []).map((c: any) => ({
   293→                name: c.name,
   294→                hex: c.hex,
   295→                stock: c.stock ?? 0,
   296→              })),
   297→              sizes: (prod.sizes ?? []).map((s: any) => ({
   298→                name: s.name,
   299→                stock: s.stock ?? 0,
   300→              })),
   301→            }))
   302→          }
   303→        }
   304→      } catch {
   305→        /* ignore */
   306→      }
   307→    }
   308→    setOpen(true)
   309→  }
   310→
   311→  const filtered = useMemo(() => {
   312→    const list = productsQuery.data?.items ?? []
   313→    if (!search.trim()) return list
   314→    const q = search.trim().toLowerCase()
   315→    return list.filter(
   316→      (p) =>
   317→        p.name.toLowerCase().includes(q) ||
   318→        p.slug.toLowerCase().includes(q) ||
   319→        p.sku?.toLowerCase().includes(q),
   320→    )
   321→  }, [productsQuery.data, search])
   322→
   323→  return (
   324→    <div className="space-y-5">
   325→      {/* HEADER */}
   326→      <div className="flex flex-col items-start justify-between gap-3 sm:flex-row sm:items-center">
   327→        <div>
   328→          <p className="text-[10px] uppercase tracking-[0.25em] text-muted-foreground">
   329→            Catálogo
   330→          </p>
   331→          <h2 className="font-serif text-2xl">Produtos</h2>
   332→        </div>
   333→        <div className="flex w-full items-center gap-2 sm:w-auto">
   334→          <Input
   335→            value={search}
   336→            onChange={(e) => setSearch(e.target.value)}
   337→            placeholder="Buscar por nome, slug ou SKU…"
   338→            className="h-10 sm:w-64"
   339→          />
   340→          <Button
   341→            onClick={newProduct}
   342→            className="btn-gold h-10 gap-2 border-0 px-4 text-xs uppercase tracking-[0.2em] hover:brightness-110"
   343→          >
   344→            <Plus className="size-4" />
   345→            Novo
   346→          </Button>
   347→        </div>
   348→      </div>
   349→
   350→      {/* TABLE */}
   351→      <div className="rounded-lg border border-border bg-surface">
   352→        {productsQuery.isLoading ? (
   353→          <div className="space-y-2 p-4">
   354→            {Array.from({ length: 5 }).map((_, i) => (
   355→              <Skeleton key={i} className="h-14 w-full" />
   356→            ))}
   357→          </div>
   358→        ) : filtered.length === 0 ? (
   359→          <div className="flex flex-col items-center justify-center gap-3 py-16 text-center">
   360→            <Folder className="size-10 text-muted-foreground/60" />
   361→            <p className="text-sm text-muted-foreground">
   362→              {search.trim()
   363→                ? 'Nenhum produto encontrado para a busca.'
   364→                : 'Ainda não há produtos cadastrados.'}
   365→            </p>
   366→            <Button
   367→              onClick={newProduct}
   368→              className="btn-gold mt-1 h-10 gap-2 border-0 text-xs uppercase tracking-[0.2em]"
   369→            >
   370→              <Plus className="size-4" />
   371→              Criar primeiro produto
   372→            </Button>
   373→          </div>
   374→        ) : (
   375→          <div className="overflow-x-auto">
   376→            <table className="w-full text-sm">
   377→              <thead>
   378→                <tr className="border-b border-border text-left text-[11px] uppercase tracking-wider text-muted-foreground">
   379→                  <th className="py-3 pl-4 pr-3">Produto</th>
   380→                  <th className="px-3 py-3 hidden md:table-cell">Categoria</th>
   381→                  <th className="px-3 py-3">Preço</th>
   382→                  <th className="px-3 py-3">Status</th>
   383→                  <th className="px-3 py-3 hidden sm:table-cell">Estoque</th>
   384→                  <th className="px-3 py-3 text-right">Ações</th>
   385→                </tr>
   386→              </thead>
   387→              <tbody>
   388→                {filtered.map((p) => {
   389→                  const img = p.images?.[0]?.url
   390→                    ? `${p.images[0].url}`
   391→                    : null
   392→                  return (
   393→                    <tr
   394→                      key={p.id}
   395→                      className="border-b border-border last:border-0 hover:bg-muted/40"
   396→                    >
   397→                      <td className="py-3 pl-4 pr-3">
   398→                        <div className="flex items-center gap-3">
   399→                          <div className="grid size-12 shrink-0 place-items-center overflow-hidden rounded-md border border-border bg-muted">
   400→                            {img ? (
   401→                              <img
   402→                                src={img}
   403→                                alt={p.name}
   404→                                className="size-full object-cover"
   405→                              />
   406→                            ) : (
   407→                              <ImagePlus className="size-4 text-muted-foreground" />
   408→                            )}
   409→                          </div>
   410→                          <div className="min-w-0">
   411→                            <p className="truncate font-medium text-foreground">
   412→                              {p.name}
   413→                            </p>
   414→                            <p className="truncate text-[11px] text-muted-foreground">
   415→                              {p.slug}
   416→                              {p.sku ? ` · ${p.sku}` : ''}
   417→                            </p>
   418→                            {p.featured && (
   419→                              <span className="mt-0.5 inline-flex items-center gap-1 text-[10px] uppercase tracking-wider text-accent">
   420→                                <Star className="size-3 fill-accent" />
   421→                                Destaque
   422→                              </span>
   423→                            )}
   424→                          </div>
   425→                        </div>
   426→                      </td>
   427→                      <td className="px-3 py-3 hidden md:table-cell">
   428→                        <span className="text-xs text-muted-foreground">
   429→                          {p.category?.name ?? '—'}
   430→                        </span>
   431→                      </td>
   432→                      <td className="px-3 py-3">
   433→                        <p className="font-medium">{formatBRL(p.price)}</p>
   434→                        {p.compareAt ? (
   435→                          <p className="text-[11px] text-muted-foreground line-through">
   436→                            {formatBRL(p.compareAt)}
   437→                          </p>
   438→                        ) : null}
   439→                      </td>
   440→                      <td className="px-3 py-3">
   441→                        <span
   442→                          className={`inline-flex rounded px-2 py-1 text-[10px] uppercase tracking-wider ${STATUS_COLORS[p.status] ?? 'bg-muted'}`}
   443→                        >
   444→                          {STATUS_LABELS[p.status] ?? p.status}
   445→                        </span>
   446→                      </td>
   447→                      <td className="px-3 py-3 hidden sm:table-cell text-xs text-muted-foreground">
   448→                        {p._count?.colors ?? 0} cores · {p._count?.sizes ?? 0}{' '}
   449→                        tamanhos
   450→                      </td>
   451→                      <td className="px-3 py-3">
   452→                        <div className="flex items-center justify-end gap-1">
   453→                          <button
   454→                            onClick={() => editProduct(p)}
   455→                            aria-label="Editar"
   456→                            className="grid size-9 place-items-center rounded-md text-foreground hover:bg-muted"
   457→                          >
   458→                            <Pencil className="size-4" />
   459→                          </button>
   460→                          <button
   461→                            onClick={() => {
   462→                              setDeleteId(p.id)
   463→                              setDeleteName(p.name)
   464→                            }}
   465→                            aria-label="Excluir"
   466→                            className="grid size-9 place-items-center rounded-md text-destructive hover:bg-destructive/10"
   467→                          >
   468→                            <Trash2 className="size-4" />
   469→                          </button>
   470→                        </div>
   471→                      </td>
   472→                    </tr>
   473→                  )
   474→                })}
   475→              </tbody>
   476→            </table>
   477→          </div>
   478→        )}
   479→      </div>
   480→
   481→      {/* EDITOR SHEET */}
   482→      <Sheet open={open} onOpenChange={setOpen}>
   483→        <SheetContent
   484→          side="right"
   485→          className="flex w-full flex-col gap-0 bg-background p-0 sm:max-w-2xl lg:max-w-3xl"
   486→        >
   487→          <SheetHeader className="flex flex-row items-center justify-between border-b border-border p-5">
   488→            <div>
   489→              <SheetTitle className="font-serif text-2xl">
   490→                {editId ? 'Editar produto' : 'Novo produto'}
   491→              </SheetTitle>
   492→              <SheetDescription className="text-xs">
   493→                {editId
   494→                  ? 'Atualize os campos abaixo e salve.'
   495→                  : 'Preencha os campos e crie um novo item no catálogo.'}
   496→              </SheetDescription>
   497→            </div>
   498→          </SheetHeader>
   499→
   500→          <div className="flex-1 overflow-y-auto p-5">
   501→            <ProductEditor
   502→              form={form}
   503→              setForm={setForm}
   504→              categories={categories}
   505→            />
   506→          </div>
   507→
   508→          <div className="flex items-center justify-end gap-3 border-t border-border bg-surface p-5">
   509→            <Button
   510→              variant="outline"
   511→              onClick={() => setOpen(false)}
   512→              className="h-11 px-6 text-xs uppercase tracking-[0.2em]"
   513→            >
   514→              Cancelar
   515→            </Button>
   516→            <Button
   517→              onClick={() => saveMutation.mutate()}
   518→              disabled={
   519→                saveMutation.isPending ||
   520→                !form.name.trim() ||
   521→                !form.categoryId ||
   522→                !form.price
   523→              }
   524→              className="btn-gold h-11 gap-2 border-0 px-6 text-xs uppercase tracking-[0.2em] disabled:opacity-50"
   525→            >
   526→              {saveMutation.isPending ? (
   527→                <Loader2 className="size-4 animate-spin" />
   528→              ) : (
   529→                <Save className="size-4" />
   530→              )}
   531→              {editId ? 'Salvar alterações' : 'Criar produto'}
   532→            </Button>
   533→          </div>
   534→        </SheetContent>
   535→      </Sheet>
   536→
   537→      {/* DELETE CONFIRM */}
   538→      <AlertDialog
   539→        open={!!deleteId}
   540→        onOpenChange={(o) => !o && setDeleteId(null)}
   541→      >
   542→        <AlertDialogContent>
   543→          <AlertDialogHeader>
   544→            <AlertDialogTitle>Excluir produto?</AlertDialogTitle>
   545→            <AlertDialogDescription>
   546→              Tem certeza que deseja remover <b>{deleteName}</b> do catálogo?
   547→              Esta ação não pode ser desfeita.
   548→            </AlertDialogDescription>
   549→          </AlertDialogHeader>
   550→          <AlertDialogFooter>
   551→            <AlertDialogCancel className="h-11">Cancelar</AlertDialogCancel>
   552→            <AlertDialogAction
   553→              onClick={() => deleteId && deleteMutation.mutate(deleteId)}
   554→              className="h-11 bg-destructive text-white hover:bg-destructive/90"
   555→            >
   556→              {deleteMutation.isPending ? (
   557→                <Loader2 className="size-4 animate-spin" />
   558→              ) : (
   559→                <Trash2 className="size-4" />
   560→              )}
   561→              Excluir
   562→            </AlertDialogAction>
   563→          </AlertDialogFooter>
   564→        </AlertDialogContent>
   565→      </AlertDialog>
   566→    </div>
   567→  )
   568→}

   570→// ───────────────────────── PRODUCT EDITOR ─────────────────────────
   571→function ProductEditor({
   572→  form,
   573→  setForm,
   574→  categories,
   575→}: {
   576→  form: ProductForm
   577→  setForm: React.Dispatch<React.SetStateAction<ProductForm>>
   578→  categories: Category[]
   579→}) {
   580→  const [uploading, setUploading] = useState(false)
   581→  const [dragOver, setDragOver] = useState(false)
   582→
   583→  // Atualiza slug automaticamente quando muda o nome (apenas se slug vazio ou igual ao slug do nome antigo)
   584→  useEffect(() => {
   585→    if (
   586→      form.name &&
   587→      (!form.slug || form.slug === slugify(form.name.slice(0, -1)))
   588→    ) {
   589→      setForm((f) => ({ ...f, slug: slugify(f.name) }))

   593→  async function uploadFiles(files: FileList | File[]) {
   594→    const arr = Array.from(files).filter((f) => f.type.startsWith('image/'))
   595→    if (arr.length === 0) return
   596→    if (form.images.length + arr.length > 20) {
   597→      toast.error('Máximo de 20 imagens por produto.')
   598→      return
   599→    }
   600→    setUploading(true)
   601→    try {
   602→      const fd = new FormData()
   603→      arr.forEach((f) => fd.append('file', f))
   604→      const r = await fetch('/api/admin/upload', { method: 'POST', body: fd })
   605→      const data = await r.json()
   606→      if (!r.ok) throw new Error(data?.error ?? 'Falha no upload')
   607→      const urls: string[] = data.urls ?? []
   608→      setForm((f) => ({
   609→        ...f,
   610→        images: [
   611→          ...f.images,
   612→          ...urls.map<ImageDraft>((url, i) => ({
   613→            url,
   614→            isMain: f.images.length === 0 && i === 0,
   615→            order: f.images.length + i,
   616→          })),
   617→        ],
   618→      }))
   619→      toast.success(`${urls.length} imagem(s) enviada(s).`)
   620→    } catch (e) {
   621→      toast.error((e as Error).message)
   622→    } finally {
   623→      setUploading(false)
   624→    }
   625→  }
